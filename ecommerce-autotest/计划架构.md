# 电商后台管理系统自动化测试 - 架构计划文档

## 1. 项目概述

### 1.1 项目背景
随着电商行业的快速发展，后台管理系统的稳定性和可靠性变得至关重要。本毕业设计项目旨在开发一套基于Selenium和Python的自动化测试系统，用于验证电商后台管理系统的核心功能，提高测试效率和质量。

### 1.2 项目目标
- 实现电商后台核心业务模块的自动化测试
- 构建可维护、可扩展的测试框架
- 提供详细的测试报告和日志分析
- 引入AI能力，对失败用例进行智能原因分析与修复建议生成
- 在测试报告中提供AI摘要、风险归纳与优化建议
- 提升测试效率，减少人工测试成本

### 1.3 项目范围
**包含模块：**
- 登录功能测试
- 商品管理测试
- 订单管理测试
- 用户管理测试
- 权限管理测试
- AI失败分析
- AI报告总结

**技术边界：**
- 前端UI自动化测试（不包含接口测试）
- 基于Web的电商后台系统（不包含移动端）
- 主流浏览器兼容性测试
- AI能力聚焦测试结果分析与报告增强，不替代核心自动化执行流程

```python
class BasePage:
    """所有页面对象的基类"""
    def __init__(self, driver):
        self.driver = driver
        self.timeout = 10
        self.wait = WebDriverWait(driver, self.timeout)
    
    def find_element(self, locator):
        """查找元素（带等待机制）"""
        return self.wait.until(
            EC.presence_of_element_located(locator)
        )
    
    def take_screenshot(self, filename):
        """页面截图"""
        self.driver.save_screenshot(filename)
```

```python
class LoginPage(BasePage):

    USERNAME_INPUT = (By.ID, 'username')
    PASSWORD_INPUT = (By.ID, 'password')
    LOGIN_BUTTON = (By.ID, 'login-btn')
    ERROR_MESSAGE = (By.CLASS_NAME, 'error-msg')
    
    def login(self, username, password):
        """执行登录操作"""
        self.find_element(self.USERNAME_INPUT).send_keys(username)
        self.find_element(self.PASSWORD_INPUT).send_keys(password)
        self.find_element(self.LOGIN_BUTTON).click()

        """验证登录是否成功"""
        # 检查是否跳转到首页
        return "dashboard" in self.driver.current_url
    
    def get_error_message(self):
        """获取错误提示信息"""
        return self.find_element(self.ERROR_MESSAGE).text
```

```python
class ConfigManager:
    """配置管理类"""
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._load_config()
        return cls._instance
    
    def _load_config(self):
        """加载配置文件"""
        config_path = os.path.join(
            os.path.dirname(__file__), 
            '../config/config.yaml'
        )
        with open(config_path, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)
    
    def get(self, key, default=None):
        """获取配置值"""
        keys = key.split('.')
        value = self.config
        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return default
        return value
    
    def set_environment(self, env):
        """动态切换环境"""
        self.config['environment']['test_env'] = env
```

### 3.3 执行引擎设计

#### 3.3.1 测试执行器
```python
class TestExecutor:
    """测试执行引擎"""
    def __init__(self, test_suite, config):
        self.test_suite = test_suite
        self.config = config
        self.results = []
        self.report_generator = ReportGenerator()
        
    def execute(self):
        """执行测试套件"""
        start_time = time.time()
        
        for test_case in self.test_suite:
            result = self._execute_test_case(test_case)
            self.results.append(result)
            
            if result.status == 'FAILED' and self.config['test']['retry_count'] > 0:
                # 失败重试逻辑
                self._retry_test_case(test_case, result)
        
        end_time = time.time()
        execution_time = end_time - start_time

        # 调用AI分析失败结果
        if self.config.get('ai', {}).get('analyze_failures', False):
            self._analyze_failures_with_ai()
        
        # 生成报告
        report_data = self._generate_report_data(execution_time)
        self.report_generator.generate(report_data)
        
        return self.results
    
    def _execute_test_case(self, test_case):
        """执行单个测试用例"""
        test_result = {
            'name': test_case.__name__,
            'start_time': time.time(),
            'status': 'PASSED',
            'error': None,
            'screenshot': None
        }
        
        try:
            test_case_instance = test_case()
            test_case_instance.setup_method()
            test_case_instance.run()
            test_case_instance.teardown_method()
        except AssertionError as e:
            test_result['status'] = 'FAILED'
            test_result['error'] = str(e)
            test_result['screenshot'] = self._take_screenshot()
        except Exception as e:
            test_result['status'] = 'ERROR'
            test_result['error'] = str(e)
            test_result['screenshot'] = self._take_screenshot()
        finally:
            test_result['end_time'] = time.time()
            
        return test_result

    def _analyze_failures_with_ai(self):
        """对失败用例进行AI分析"""
        ai_service = AIAnalysisService(self.config)
        for result in self.results:
            if result['status'] in ['FAILED', 'ERROR']:
                result['ai_analysis'] = ai_service.analyze_failure(result)
```

#### 3.3.2 并行执行支持
```python
class ParallelExecutor:
    """并行测试执行器"""
    def __init__(self, max_workers=5):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        
    def execute_parallel(self, test_suites):
        """并行执行多个测试套件"""
        futures = []
        for suite in test_suites:
            future = self.executor.submit(self._execute_suite, suite)
            futures.append(future)
        
        results = []
        for future in as_completed(futures):
            results.extend(future.result())
        
        return results
```

#### 3.4.1 报告数据结构
```python
class ReportData:
    """报告数据结构"""
    def __init__(self):
        self.project_info = {}
        self.execution_info = {}
        self.test_results = []
        self.statistics = {}
        
    def calculate_statistics(self):
        """计算统计信息"""
        total = len(self.test_results)
        passed = len([r for r in self.test_results if r['status'] == 'PASSED'])
        failed = len([r for r in self.test_results if r['status'] == 'FAILED'])
        error = len([r for r in self.test_results if r['status'] == 'ERROR'])
        
        self.statistics = {
            'total': total,
            'passed': passed,
            'failed': failed,
            'error': error,
            'pass_rate': (passed / total * 100) if total > 0 else 0,
            'execution_time': self.execution_info.get('total_time', 0)
        }
```

#### 3.4.2 HTML报告生成器
```python
class HTMLReportGenerator:
    """HTML报告生成器"""
    def generate(self, report_data, template='default'):
        """生成HTML报告"""
        template_path = self._get_template_path(template)

        if self.config.get('ai.generate_summary', False):
            report_data.ai_summary = self._build_ai_summary(report_data)

        html_content = self._render_template(template_path, report_data)
        
        report_dir = self._ensure_report_dir()
        report_file = os.path.join(report_dir, self._generate_filename())
        
        with open(report_file, 'w', encoding='utf-8') as f:
            f.write(html_content)
        
        # 如果需要，发送邮件
        if self.config.get('report.send_email', False):
            self._send_email(report_file)
        
        return report_file
    
    def _render_template(self, template_path, data):
        """渲染模板"""
        with open(template_path, 'r', encoding='utf-8') as f:
            template = Template(f.read())
        
        return template.render(
            project=data.project_info,
            execution=data.execution_info,
            results=data.test_results,
            stats=data.statistics,
            ai_summary=getattr(data, 'ai_summary', {}),
        )
```
